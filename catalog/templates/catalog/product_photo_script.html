(() => { const urls = new WeakMap(); document.addEventListener('change', event => { const input = event.target; if (!input.matches('input[type="file"][name^="photos-"]')) return; const container = input.closest('.inline-related'); if (!container) return; let preview = container.querySelector('[data-product-upload-preview]'); if (!preview) { preview = document.createElement('img'); preview.className = 'product-upload-preview'; preview.dataset.productUploadPreview = ''; preview.alt = 'Selected product photo preview'; input.closest('.form-row').append(preview); } if (urls.has(input)) URL.revokeObjectURL(urls.get(input)); const file = input.files[0]; preview.hidden = !file; if (!file) { preview.removeAttribute('src'); return; } const url = URL.createObjectURL(file); urls.set(input, url); preview.src = url; }); })();
